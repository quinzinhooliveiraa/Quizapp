## 1. Backend — novo endpoint de checagem de admin

Criar `artifacts/api-server/src/routes/admin.ts`:

```ts
import { Router } from "express";
import { eq } from "drizzle-orm";
import { db, sessionsTable } from "@workspace/db";

const router = Router();

function getAdminEmails(): string[] {
  return (process.env.ADMIN_EMAILS || "")
    .split(",")
    .map((v) => v.trim().toLowerCase())
    .filter(Boolean);
}

// GET /api/admin/check?sessionId=...
router.get("/check", async (req, res) => {
  const sessionId = typeof req.query.sessionId === "string" ? req.query.sessionId : undefined;
  if (!sessionId) return res.json({ isAdmin: false });

  const [row] = await db
    .select({ buyerEmail: sessionsTable.buyerEmail })
    .from(sessionsTable)
    .where(eq(sessionsTable.id, sessionId))
    .limit(1);

  if (!row?.buyerEmail) return res.json({ isAdmin: false });

  const isAdmin = getAdminEmails().includes(row.buyerEmail.trim().toLowerCase());
  return res.json({ isAdmin });
});

export const adminRouter = router;
```

Registrar em `artifacts/api-server/src/routes/index.ts` (junto com os outros):
```ts
import { adminRouter } from "./admin.js";
// ...
router.use("/admin", adminRouter);
```

## 2. Frontend — nova página `artifacts/perguntas-de-conexao/src/pages/Admin.tsx`

```tsx
import { useEffect, useState } from 'react';
import { Link } from 'wouter';
import { ArrowLeft, ExternalLink, Copy, Check, ShieldAlert, LayoutTemplate } from 'lucide-react';

function safeGetItem(key: string): string | null {
  try { return localStorage.getItem(key); } catch { return null; }
}

type LandingEntry = {
  id: string;
  label: string;
  path: string;
  description: string;
};

// Lista de landing pages ativas. Adicionar novas aqui conforme criadas.
const LANDINGS: LandingEntry[] = [
  { id: 'v1', label: 'Landing V1 — Estática', path: '/', description: 'Hero + dor + solução + temas + prova social + preço + FAQ, tudo em scroll.' },
  { id: 'v2', label: 'Landing V2 — Quiz interativo', path: '/lp2', description: 'Mesma base, mas com mini-quiz de 3 perguntas antes de mostrar o preço.' },
];

export default function Admin() {
  const [status, setStatus] = useState<'checking' | 'denied' | 'ok'>('checking');
  const [copiedId, setCopiedId] = useState<string | null>(null);

  useEffect(() => {
    const sessionId = safeGetItem('conexao-session');
    if (!sessionId) { setStatus('denied'); return; }
    const apiBase = (import.meta.env.VITE_API_URL || '').replace(/\/+$/, '');
    fetch(`${apiBase}/api/admin/check?sessionId=${encodeURIComponent(sessionId)}`)
      .then((r) => (r.ok ? r.json() : { isAdmin: false }))
      .then((data) => setStatus(data.isAdmin ? 'ok' : 'denied'))
      .catch(() => setStatus('denied'));
  }, []);

  const origin = typeof window !== 'undefined' ? window.location.origin : '';

  const copyLink = (entry: LandingEntry) => {
    const url = `${origin}${entry.path}`;
    navigator.clipboard?.writeText(url).then(() => {
      setCopiedId(entry.id);
      setTimeout(() => setCopiedId((current) => (current === entry.id ? null : current)), 1800);
    });
  };

  if (status === 'checking') {
    return <div className="admin-page admin-loading"><p>Verificando acesso…</p></div>;
  }

  if (status === 'denied') {
    return (
      <div className="admin-page admin-denied">
        <ShieldAlert size={32} />
        <h1>Acesso restrito</h1>
        <p>Esta área é reservada. Se você acha que deveria ter acesso, entre com sua conta primeiro.</p>
        <Link href="/login" className="button button-primary">Ir para o login</Link>
      </div>
    );
  }

  return (
    <div className="admin-page">
      <header className="admin-header">
        <Link href="/app" className="admin-back"><ArrowLeft size={16} /> Voltar ao app</Link>
        <h1><LayoutTemplate size={22} /> Painel Admin</h1>
        <p>Links das landing pages ativas. Estatísticas de conversão entram numa próxima etapa.</p>
      </header>

      <section className="admin-section">
        <h2>Landing pages</h2>
        <div className="admin-landing-grid">
          {LANDINGS.map((entry) => (
            <article key={entry.id} className="admin-landing-card" data-testid={`card-landing-${entry.id}`}>
              <div className="admin-landing-info">
                <strong>{entry.label}</strong>
                <p>{entry.description}</p>
                <code className="admin-landing-url">{origin}{entry.path}</code>
              </div>
              <div className="admin-landing-actions">
                <a
                  href={`${origin}${entry.path}`}
                  target="_blank"
                  rel="noopener noreferrer"
                  className="admin-icon-button"
                  data-testid={`link-open-landing-${entry.id}`}
                  aria-label={`Abrir ${entry.label}`}
                >
                  <ExternalLink size={16} />
                </a>
                <button
                  type="button"
                  onClick={() => copyLink(entry)}
                  className="admin-icon-button"
                  data-testid={`button-copy-landing-${entry.id}`}
                  aria-label={`Copiar link de ${entry.label}`}
                >
                  {copiedId === entry.id ? <Check size={16} /> : <Copy size={16} />}
                </button>
              </div>
            </article>
          ))}
        </div>
      </section>
    </div>
  );
}
```
