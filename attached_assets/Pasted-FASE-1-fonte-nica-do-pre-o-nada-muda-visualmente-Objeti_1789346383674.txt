FASE 1 — fonte única do preço (nada muda visualmente)

Objetivo: tirar o preço de 10 lugares hardcoded e passar tudo a ler de um só lugar, no servidor. No fim desta fase o site continua exatamente igual, em reais.

1.1 Criar o módulo de preço no backend

Criar artifacts/api-server/src/lib/pricing.ts:

ts
export type PricingRegion = "BR" | "PT";

export type Pricing = {
  region: PricingRegion;
  currency: "brl" | "eur";
  amountCents: number;
  display: string;
  unitNote: string;
  pixAvailable: boolean;
};

const PRICING: Record<PricingRegion, Pricing> = {
  BR: {
    region: "BR",
    currency: "brl",
    amountCents: 4790,
    display: "R$ 47,90",
    unitNote: "dá 10 centavos por noite",
    pixAvailable: true,
  },
  PT: {
    region: "PT",
    currency: "eur",
    amountCents: 1490,
    display: "14,90 €",
    unitNote: "dá 3 cêntimos por noite",
    pixAvailable: false,
  },
};

export function getPricing(region: PricingRegion): Pricing {
  return PRICING[region];
}

/**
 * Resolve a região SEMPRE a partir do request, nunca do corpo enviado
 * pelo cliente. Retorna "BR" em caso de dúvida.
 */
export function resolveRegion(req: {
  headers: Record<string, unknown>;
  query?: Record<string, unknown>;
}): PricingRegion {
  // 1. Override explícito do usuário (só escolhe entre as regiões fixas,
  //    nunca um valor arbitrário — seguro por construção).
  const forced = String(req.query?.regiao ?? "").toUpperCase();
  if (forced === "PT" || forced === "BR") return forced;

  // 2. Header de país do proxy, se existir neste ambiente.
  const country = String(
    req.headers["cf-ipcountry"] ??
      req.headers["x-vercel-ip-country"] ??
      req.headers["x-country"] ??
      "",
  ).toUpperCase();
  if (country === "PT") return "PT";
  if (country === "BR") return "BR";

  // 3. Idioma do navegador.
  const lang = String(req.headers["accept-language"] ?? "").toLowerCase();
  if (lang.includes("pt-pt")) return "PT";

  // 4. Padrão.
  return "BR";
}

1.2 Expor num endpoint

Em artifacts/api-server/src/routes/, adicionar uma rota pública:

GET /api/pricing

que responde getPricing(resolveRegion(req)) como JSON. Sem autenticação, sem cache (ou cache curto), porque a resposta varia por visitante.
