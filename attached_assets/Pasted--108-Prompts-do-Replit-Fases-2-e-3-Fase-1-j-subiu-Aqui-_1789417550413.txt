```ts
export type Lp1Diagnosis = Lp3Narrative & { title: string; ponte?: string };

export function selectLp1Diagnosis(answers: Lp1QuizAnswers): Lp1Diagnosis {
  const narrative = selectLp3Narrative(toLp3Answers(answers));
  const p = answers.pain ? PAIN_PITCH[answers.pain] : undefined;
  if (p) {
    return { ...narrative, title: p.title, insight: p.insight, ponte: p.ponte,
             themeId: p.themeId, personalizations: [] };
  }
  return { ...narrative, title: diagnosisTitles[narrative.narrativeType] };
}
```
