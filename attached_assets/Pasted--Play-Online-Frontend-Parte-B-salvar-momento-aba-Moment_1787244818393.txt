E adicionar state + função `saveMoment` no topo do componente (perto de outros useState):

```tsx
const [savedIds, setSavedIds] = useState<Set<string>>(new Set());
const apiBase = (import.meta.env.VITE_API_URL || '').replace(/\/+$/, '');
const sessionIdLS = safeGet('conexao-session');
const guestTokenLS = safeGet('conexao-guest-token');

const saveMoment = async (answer: { playerId: string; playerName: string; answer: string }) => {
  if (!currentQuestion || !state.themeId) return;
  const key = answer.playerId + ':' + currentQuestion.id;
  if (savedIds.has(key)) return;
  try {
    await fetch(`${apiBase}/api/moments`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        sessionId: sessionIdLS,
        guestToken: guestTokenLS,
        questionId: currentQuestion.id,
        themeId: state.themeId,
        fromPlayerName: answer.playerName,
        answerText: answer.answer,
        roomCode: state.code,
      }),
    });
    setSavedIds(prev => { const next = new Set(prev); next.add(key); return next; });
  } catch { /* silently fail */ }
};
```

E adicionar state + fetch dos momentos no componente `AppExperienceReference` (perto de outros useState):

```tsx
type SavedMoment = {
  id: string;
  questionId: string;
  themeId: string;
  fromPlayerName: string;
  answerText: string;
  roomCode: string | null;
  createdAt: string;
};
const [savedMoments, setSavedMoments] = useState<SavedMoment[]>([]);

useEffect(() => {
  const sid = safeGetItem('conexao-session');
  const tok = safeGetItem('conexao-guest-token');
  if (!sid && !tok) return;
  const apiBase = (import.meta.env.VITE_API_URL || '').replace(/\/+$/, '');
  const qs = sid ? `sessionId=${encodeURIComponent(sid)}` : `guestToken=${encodeURIComponent(tok || '')}`;
  fetch(`${apiBase}/api/moments?${qs}`)
    .then(r => r.ok ? r.json() : null)
    .then(data => { if (data?.moments) setSavedMoments(data.moments); })
    .catch(() => { /* noop */ });
}, [activeNav]); // Refetch quando muda de aba (entrar em "Meu espaço" traz o estado atualizado)

const deleteMoment = async (id: string) => {
  const sid = safeGetItem('conexao-session');
  const tok = safeGetItem('conexao-guest-token');
  const apiBase = (import.meta.env.VITE_API_URL || '').replace(/\/+$/, '');
  const qs = sid ? `sessionId=${encodeURIComponent(sid)}` : `guestToken=${encodeURIComponent(tok || '')}`;
  await fetch(`${apiBase}/api/moments/${encodeURIComponent(id)}?${qs}`, { method: 'DELETE' }).catch(() => { /* noop */ });
  setSavedMoments(prev => prev.filter(m => m.id !== id));
};
```
