**Contexto:** hoje quem perde o localStorage perde acesso ao baralho (mesmo já tendo comprado). Fix: pessoa digita email, recebe código de 6 dígitos, valida, sessão restaurada. `buyerEmail` já existe na tabela `sessions` (quem paga pelo Abacate já fica com email salvo), então basta criar a tabela de códigos + endpoints + integração Brevo.

## 2. Criar integração Brevo — `artifacts/api-server/src/lib/brevo.ts` (novo arquivo)

```ts
const BREVO_API_URL = "https://api.brevo.com/v3/smtp/email";

type SendEmailParams = {
  to: string;
  toName?: string;
  subject: string;
  htmlContent: string;
  textContent?: string;
};

export async function sendEmailViaBrevo(params: SendEmailParams): Promise<{ ok: boolean; error?: string }> {
  const apiKey = process.env.BREVO_API_KEY;
  const fromEmail = process.env.EMAIL_FROM;
  const fromName = process.env.EMAIL_FROM_NAME || "Perguntas de Conexão";

  if (!apiKey || !fromEmail) {
    console.error("[brevo] Missing BREVO_API_KEY or EMAIL_FROM env var");
    return { ok: false, error: "email not configured" };
  }

  try {
    const response = await fetch(BREVO_API_URL, {
      method: "POST",
      headers: {
        "accept": "application/json",
        "api-key": apiKey,
        "content-type": "application/json",
      },
      body: JSON.stringify({
        sender: { email: fromEmail, name: fromName },
        to: [{ email: params.to, name: params.toName || params.to }],
        subject: params.subject,
        htmlContent: params.htmlContent,
        textContent: params.textContent,
      }),
    });

    if (!response.ok) {
      const body = await response.text();
      console.error(`[brevo] send failed ${response.status}:`, body);
      return { ok: false, error: `brevo ${response.status}` };
    }
    return { ok: true };
  } catch (err) {
    console.error("[brevo] send exception:", err);
    return { ok: false, error: String(err) };
  }
}

export function buildLoginCodeEmail(code: string) {
  const subject = `Seu código de acesso: ${code}`;
  const htmlContent = `
    <div style="font-family: -apple-system, sans-serif; max-width: 480px; margin: 0 auto; padding: 32px 24px; color: #17121b;">
      <p style="font-size: 14px; color: #6b6070; letter-spacing: 0.08em; text-transform: uppercase; margin: 0 0 8px;">Perguntas de Conexão</p>
      <h1 style="font-size: 22px; font-weight: 500; margin: 0 0 24px;">Seu código de acesso</h1>
      <p style="font-size: 15px; line-height: 1.55; color: #4a4550; margin: 0 0 20px;">Use o código abaixo para entrar no seu baralho. Ele expira em 15 minutos.</p>
      <div style="background: #f4f0f8; border-radius: 12px; padding: 24px; text-align: center; letter-spacing: 0.5em; font-size: 32px; font-weight: 600; color: #17121b; margin: 0 0 20px;">${code}</div>
      <p style="font-size: 13px; line-height: 1.55; color: #8b8290; margin: 0;">Se você não solicitou este código, pode ignorar este email.</p>
    </div>
  `;
  const textContent = `Perguntas de Conexão\n\nSeu código de acesso: ${code}\n\nEle expira em 15 minutos.\n\nSe você não solicitou este código, pode ignorar este email.`;
  return { subject, htmlContent, textContent };
}
```

## 3. Criar rotas de auth — `artifacts/api-server/src/routes/auth.ts` (novo arquivo)

```ts
import { Router, type IRouter } from "express";
import crypto from "node:crypto";
import { and, desc, eq, gt, lt, sql } from "drizzle-orm";
import { db, authCodesTable, sessionsTable } from "@workspace/db";
import { sendEmailViaBrevo, buildLoginCodeEmail } from "../lib/brevo";

const router: IRouter = Router();

const CODE_TTL_MINUTES = 15;
const REQUEST_COOLDOWN_SECONDS = 60;
const MAX_ATTEMPTS = 5;

function normalizeEmail(email: string): string {
  return email.trim().toLowerCase();
}

function generateCode(): string {
  return String(crypto.randomInt(0, 1_000_000)).padStart(6, "0");
}

// POST /auth/request-code — pede código de acesso pro email
router.post("/auth/request-code", async (req, res): Promise<void> => {
  const email = normalizeEmail(String(req.body?.email || ""));
  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    res.status(400).json({ error: "Email inválido" });
    return;
  }

  // Rate limit básico: se pediu código pra esse email há menos de 60s, bloqueia
  const cooldownAgo = new Date(Date.now() - REQUEST_COOLDOWN_SECONDS * 1000);
  const [recent] = await db.select({ createdAt: authCodesTable.createdAt })
    .from(authCodesTable)
    .where(and(eq(authCodesTable.email, email), gt(authCodesTable.createdAt, cooldownAgo)))
    .orderBy(desc(authCodesTable.createdAt))
    .limit(1);
  if (recent) {
    res.status(429).json({ error: "Aguarde um instante antes de pedir outro código" });
    return;
  }

  // Limpa códigos expirados do email (housekeeping)
  await db.delete(authCodesTable).where(and(
    eq(authCodesTable.email, email),
    lt(authCodesTable.expiresAt, new Date()),
  ));

  // IMPORTANTE de segurança: sempre responde OK, mesmo se o email não existe
  // (evita que atacante descubra quais emails estão cadastrados)
  const [existingSession] = await db.select({ id: sessionsTable.id })
    .from(sessionsTable)
    .where(and(eq(sessionsTable.buyerEmail, email), eq(sessionsTable.accessGranted, true)))
    .limit(1);

  if (!existingSession) {
    // Não tem conta com esse email — responde OK mesmo assim, mas não manda email
    res.json({ ok: true });
    return;
  }

  // Cria o código
  const code = generateCode();
  const id = crypto.randomUUID();
  const expiresAt = new Date(Date.now() + CODE_TTL_MINUTES * 60 * 1000);
  await db.insert(authCodesTable).values({ id, email, code, expiresAt });

  // Envia por email (Brevo)
  const emailPayload = buildLoginCodeEmail(code);
  const result = await sendEmailViaBrevo({
    to: email,
    subject: emailPayload.subject,
    htmlContent: emailPayload.htmlContent,
    textContent: emailPayload.textContent,
  });

  if (!result.ok) {
    console.error("[auth/request-code] email failed:", result.error);
    // Ainda responde OK pro cliente (não vazar detalhe), mas loga
  }

  res.json({ ok: true });
});

// POST /auth/verify-code — valida código, retorna sessões do email
router.post("/auth/verify-code", async (req, res): Promise<void> => {
  const email = normalizeEmail(String(req.body?.email || ""));
  const code = String(req.body?.code || "").trim();

  if (!email || !code || code.length !== 6) {
    res.status(400).json({ error: "Email e código de 6 dígitos são obrigatórios" });
    return;
  }

  // Busca código válido (não usado, não expirado)
  const [authCode] = await db.select()
    .from(authCodesTable)
    .where(and(
      eq(authCodesTable.email, email),
      eq(authCodesTable.code, code),
      gt(authCodesTable.expiresAt, new Date()),
    ))
    .orderBy(desc(authCodesTable.createdAt))
    .limit(1);

  if (!authCode) {
    // Pode ser: código errado, expirado, ou email não tem código
    // Incrementa tentativas em códigos ativos desse email (rate limit contra brute force)
    await db.update(authCodesTable)
      .set({ attempts: sql`${authCodesTable.attempts} + 1` })
      .where(and(
        eq(authCodesTable.email, email),
        gt(authCodesTable.expiresAt, new Date()),
      ));
    res.status(401).json({ error: "Código inválido ou expirado" });
    return;
  }

  if (authCode.usedAt) {
    res.status(401).json({ error: "Este código já foi usado" });
    return;
  }

  if (authCode.attempts >= MAX_ATTEMPTS) {
    res.status(429).json({ error: "Muitas tentativas. Peça um novo código." });
    return;
  }

  // Marca o código como usado
  await db.update(authCodesTable)
    .set({ usedAt: new Date() })
    .where(eq(authCodesTable.id, authCode.id));

  // Busca todas as sessões pagas desse email
  const sessions = await db.select({
    id: sessionsTable.id,
    buyerName: sessionsTable.buyerName,
    packageName: sessionsTable.packageName,
    createdAt: sessionsTable.createdAt,
  })
  .from(sessionsTable)
  .where(and(eq(sessionsTable.buyerEmail, email), eq(sessionsTable.accessGranted, true)))
  .orderBy(desc(sessionsTable.createdAt));

  if (sessions.length === 0) {
    res.status(404).json({ error: "Nenhum acesso encontrado para este email" });
    return;
  }

  res.json({ sessions });
});

export default router;
```

## Notas de segurança
- Códigos expiram em 15 min
- Máximo 5 tentativas por código
- Rate limit: 1 pedido de código por email a cada 60s
- Códigos são one-time-use
- Se email não existe no sistema, ainda respondemos "ok" (não vazamos quais emails estão cadastrados)
