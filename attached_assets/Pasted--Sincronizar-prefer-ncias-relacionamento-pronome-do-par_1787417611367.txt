## 2. Criar endpoint de preferências — novo arquivo `artifacts/api-server/src/routes/preferences.ts`

```ts
import { Router } from "express";
import { z } from "zod";
import { db } from "@workspace/db";
import { sessionsTable, invitesTable } from "@workspace/db/schema";
import { eq } from "drizzle-orm";

const router = Router();

const patchSchema = z.object({
  sessionId: z.string().optional(),
  guestToken: z.string().optional(),
  relationshipType: z.string().max(80).optional(),
  partnerPronoun: z.string().max(40).optional(),
});

// GET /api/preferences?sessionId=... | ?guestToken=...
router.get("/", async (req, res) => {
  const sessionId = typeof req.query.sessionId === "string" ? req.query.sessionId : undefined;
  const guestToken = typeof req.query.guestToken === "string" ? req.query.guestToken : undefined;

  if (!sessionId && !guestToken) {
    return res.status(400).json({ error: "sessionId or guestToken required" });
  }

  if (sessionId) {
    const [row] = await db
      .select({
        relationshipType: sessionsTable.relationshipType,
        partnerPronoun: sessionsTable.partnerPronoun,
      })
      .from(sessionsTable)
      .where(eq(sessionsTable.id, sessionId))
      .limit(1);
    if (!row) return res.status(404).json({ error: "session not found" });
    return res.json({
      relationshipType: row.relationshipType || null,
      partnerPronoun: row.partnerPronoun || null,
    });
  }

  const [row] = await db
    .select({
      relationshipType: invitesTable.relationshipType,
      partnerPronoun: invitesTable.partnerPronoun,
    })
    .from(invitesTable)
    .where(eq(invitesTable.token, guestToken!))
    .limit(1);
  if (!row) return res.status(404).json({ error: "invite not found" });
  return res.json({
    relationshipType: row.relationshipType || null,
    partnerPronoun: row.partnerPronoun || null,
  });
});

// PATCH /api/preferences  body: { sessionId | guestToken, relationshipType?, partnerPronoun? }
router.patch("/", async (req, res) => {
  const parsed = patchSchema.safeParse(req.body);
  if (!parsed.success) return res.status(400).json({ error: "invalid body", details: parsed.error.flatten() });
  const { sessionId, guestToken, relationshipType, partnerPronoun } = parsed.data;

  if (!sessionId && !guestToken) return res.status(400).json({ error: "sessionId or guestToken required" });

  const patch: Record<string, string | null> = {};
  if (relationshipType !== undefined) patch.relationshipType = relationshipType || null;
  if (partnerPronoun !== undefined) patch.partnerPronoun = partnerPronoun || null;
  if (Object.keys(patch).length === 0) return res.json({ ok: true });

  if (sessionId) {
    await db.update(sessionsTable).set(patch).where(eq(sessionsTable.id, sessionId));
  } else {
    await db.update(invitesTable).set(patch).where(eq(invitesTable.token, guestToken!));
  }

  return res.json({ ok: true });
});

export const preferencesRouter = router;
```

## 4. Frontend — `artifacts/perguntas-de-conexao/src/App.tsx`

### 4.1. Adicionar helper de sync (perto dos outros helpers no topo do arquivo, ou dentro do componente `AppExperienceReference`):

```tsx
const apiBase = (import.meta.env.VITE_API_URL || '').replace(/\/+$/, '');

async function fetchPreferences(sessionId: string | null, guestToken: string | null) {
  if (!sessionId && !guestToken) return null;
  const qs = sessionId ? `sessionId=${encodeURIComponent(sessionId)}` : `guestToken=${encodeURIComponent(guestToken!)}`;
  try {
    const r = await fetch(`${apiBase}/api/preferences?${qs}`);
    if (!r.ok) return null;
    return await r.json() as { relationshipType: string | null; partnerPronoun: string | null };
  } catch { return null; }
}

async function patchPreferences(
  sessionId: string | null,
  guestToken: string | null,
  patch: { relationshipType?: string; partnerPronoun?: string },
) {
  if (!sessionId && !guestToken) return;
  try {
    await fetch(`${apiBase}/api/preferences`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sessionId, guestToken, ...patch }),
    });
  } catch { /* noop */ }
}
```

### 4.2. Hidratar do servidor ao carregar (adicionar useEffect no componente `AppExperienceReference` perto dos outros useEffect):

```tsx
useEffect(() => {
  const sid = safeGetItem('conexao-session');
  const tok = safeGetItem('conexao-guest-token');
  if (!sid && !tok) return;
  fetchPreferences(sid, tok).then(prefs => {
    if (!prefs) return;
    if (prefs.relationshipType) {
      safeSetItem('conexao-relationship', prefs.relationshipType);
      setEditRelationship(prefs.relationshipType);

    if (prefs.partnerPronoun) {
      safeSetItem('conexao-partner-pronoun', prefs.partnerPronoun);
      setEditPronoun(prefs.partnerPronoun);
    }
  });
}, []);
```
